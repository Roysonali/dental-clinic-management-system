interface DataTableToolbarProps {

  children?: ReactNode;

  primaryActions?: ReactNode;
  /** Additional classes */
  className?: string;
}
 
/**

 *   columnVisibility={visibility}
 *   onColumnVisibilityChange={setVisibility}
 *   primaryActions={<Button size="md">Register Patient</Button>}

 * </DataTableToolbar>
 * ```
